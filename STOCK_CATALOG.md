import inventory from './stock_inventory.json';

export default function StockGallery() {
  return (
    <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
      {inventory.map((item, idx) => (
        <div key={idx} className="card shadow-lg rounded-xl overflow-hidden">
          <img 
            src={`/images/${item.filename}`} 
            alt={item.title} 
            className="w-full h-80 object-cover"
          />
          <div className="p-4">
            <h3 className="font-bold text-lg">{item.title}</h3>
            <p className="text-sm text-gray-600 line-clamp-2">{item.caption}</p>
            <span className="text-xs bg-gray-100 px-2 py-1 rounded mt-2 inline-block">
              {item.resolution}
            </span>
          </div>
        </div>
      ))}
    </div>
  );
}
